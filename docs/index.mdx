import Text "mo:core/Text";
import Nat "mo:core/Nat";
import Int "mo:core/Int";
import Time "mo:core/Time";
import Array "mo:core/Array";

persistent actor AuditLedger {

  public type Event = {
    id        : Nat;
    action    : Text;
    timestamp : Int;
  };

  var events : [Event] = [];
  var nextId : Nat = 0;

  public func log(action : Text) : async Nat {
    let event : Event = {
      id        = nextId;
      action    = action;
      timestamp = Time.now();
    };
    events := Array.append(events, [event]);
    nextId += 1;
    nextId - 1
  };

  public query func getEvents() : async [Event] { events };
}import Nat "mo:core/Nat";

## Deployment Script

`deployment/ic/deploy-ic.sh`

```bash
#!/bin/bash

set -e

echo "Building dashboard"
cd apps/dashboard
npm install
npm run build
cd ../..

echo "Deploying canisters"

dfx deploy mecca_token --network ic
dfx deploy treasury --network ic
dfx deploy governance --network ic
dfx deploy rewards --network ic
dfx deploy audit_ledger --network ic
dfx deploy analytics --network ic
dfx deploy marketplace --network ic
dfx deploy credentials --network ic
dfx deploy nft_registry --network ic
dfx deploy dashboard --network ic

echo "Deployment Complete"
```
